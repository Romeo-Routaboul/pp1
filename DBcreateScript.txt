CREATE DATABASE IF NOT EXISTS wishlisturquiza
    CHARACTER SET utf8mb4
    COLLATE utf8mb4_unicode_ci;

USE wishlisturquiza;

CREATE TABLE IF NOT EXISTS events (
    id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
    name VARCHAR(120) NOT NULL,
    PRIMARY KEY (id)
) ENGINE=InnoDB;

CREATE TABLE IF NOT EXISTS users (
    id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
    username VARCHAR(254) NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    role ENUM('admin', 'invitado', 'anfitrion') NOT NULL,
    event_id BIGINT UNSIGNED NULL,
    first_name VARCHAR(80) NULL,
    last_name VARCHAR(80) NULL,
    PRIMARY KEY (id),
    UNIQUE KEY uq_users_username (username),
    KEY ix_users_event_role (event_id, role),
    CONSTRAINT fk_users_event FOREIGN KEY (event_id)
        REFERENCES events (id) ON DELETE SET NULL
) ENGINE=InnoDB;

CREATE TABLE IF NOT EXISTS event_guest_profiles (
    event_id BIGINT UNSIGNED NOT NULL,
    user_id BIGINT UNSIGNED NOT NULL,
    attendance ENUM('si', 'no') NULL,
    restrictions JSON NOT NULL,
    other_restriction VARCHAR(500) NOT NULL DEFAULT '',
    PRIMARY KEY (event_id, user_id),
    CONSTRAINT fk_guest_profiles_event FOREIGN KEY (event_id)
        REFERENCES events (id) ON DELETE CASCADE,
    CONSTRAINT fk_guest_profiles_user FOREIGN KEY (user_id)
        REFERENCES users (id) ON DELETE CASCADE
) ENGINE=InnoDB;

CREATE TABLE IF NOT EXISTS minors (
    id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
    event_id BIGINT UNSIGNED NOT NULL,
    name VARCHAR(120) NOT NULL,
    age TINYINT UNSIGNED NOT NULL,
    attendance ENUM('si', 'no') NULL,
    restrictions JSON NOT NULL,
    other_restriction VARCHAR(500) NOT NULL DEFAULT '',
    PRIMARY KEY (id),
    UNIQUE KEY uq_minors_event_id (event_id, id),
    CONSTRAINT fk_minors_event FOREIGN KEY (event_id)
        REFERENCES events (id) ON DELETE CASCADE,
    CONSTRAINT ck_minors_age CHECK (age <= 17)
) ENGINE=InnoDB;

CREATE TABLE IF NOT EXISTS minor_adults (
    event_id BIGINT UNSIGNED NOT NULL,
    minor_id BIGINT UNSIGNED NOT NULL,
    user_id BIGINT UNSIGNED NOT NULL,
    PRIMARY KEY (minor_id, user_id),
    KEY ix_minor_adults_event_user (event_id, user_id),
    UNIQUE KEY uq_minor_adults_event_minor_user (event_id, minor_id, user_id),
    CONSTRAINT fk_minor_adults_minor FOREIGN KEY (event_id, minor_id)
        REFERENCES minors (event_id, id) ON DELETE CASCADE,
    CONSTRAINT fk_minor_adults_guest_profile FOREIGN KEY (event_id, user_id)
        REFERENCES event_guest_profiles (event_id, user_id) ON DELETE CASCADE
) ENGINE=InnoDB;

CREATE TABLE IF NOT EXISTS gifts (
    id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
    event_id BIGINT UNSIGNED NOT NULL,
    name VARCHAR(120) NOT NULL,
    description VARCHAR(1000) NOT NULL,
    image_url TEXT NULL,
    purchase_url TEXT NOT NULL,
    reserved_by_user_id BIGINT UNSIGNED NULL,
    PRIMARY KEY (id),
    UNIQUE KEY uq_gifts_event_id (event_id, id),
    CONSTRAINT fk_gifts_event FOREIGN KEY (event_id)
        REFERENCES events (id) ON DELETE CASCADE,
    CONSTRAINT fk_gifts_reserved_by FOREIGN KEY (reserved_by_user_id)
        REFERENCES users (id) ON DELETE SET NULL
) ENGINE=InnoDB;

CREATE TABLE IF NOT EXISTS event_tables (
    id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
    event_id BIGINT UNSIGNED NOT NULL,
    name VARCHAR(80) NOT NULL,
    seat_count SMALLINT UNSIGNED NOT NULL,
    PRIMARY KEY (id),
    UNIQUE KEY uq_event_tables_event_id (event_id, id),
    CONSTRAINT fk_event_tables_event FOREIGN KEY (event_id)
        REFERENCES events (id) ON DELETE CASCADE,
    CONSTRAINT ck_event_tables_seat_count CHECK (seat_count BETWEEN 1 AND 100)
) ENGINE=InnoDB;

CREATE TABLE IF NOT EXISTS table_assignments (
    id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
    event_id BIGINT UNSIGNED NOT NULL,
    table_id BIGINT UNSIGNED NOT NULL,
    seat_number SMALLINT UNSIGNED NOT NULL,
    guest_id BIGINT UNSIGNED NULL,
    minor_id BIGINT UNSIGNED NULL,
    minor_adult_id BIGINT UNSIGNED NULL,
    PRIMARY KEY (id),
    UNIQUE KEY uq_table_assignments_seat (table_id, seat_number),
    UNIQUE KEY uq_table_assignments_guest (event_id, guest_id),
    UNIQUE KEY uq_table_assignments_minor (event_id, minor_id),
    CONSTRAINT fk_table_assignments_table FOREIGN KEY (event_id, table_id)
        REFERENCES event_tables (event_id, id) ON DELETE CASCADE,
    CONSTRAINT fk_table_assignments_guest FOREIGN KEY (event_id, guest_id)
        REFERENCES event_guest_profiles (event_id, user_id) ON DELETE CASCADE,
    CONSTRAINT fk_table_assignments_minor_adult FOREIGN KEY (event_id, minor_id, minor_adult_id)
        REFERENCES minor_adults (event_id, minor_id, user_id) ON DELETE CASCADE,
    CONSTRAINT ck_table_assignments_occupant CHECK (
        (guest_id IS NOT NULL AND minor_id IS NULL AND minor_adult_id IS NULL)
        OR
        (guest_id IS NULL AND minor_id IS NOT NULL AND minor_adult_id IS NOT NULL)
    )
) ENGINE=InnoDB;
